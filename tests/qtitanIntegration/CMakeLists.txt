cmake_minimum_required(VERSION 3.21)
project(HyperCommanderQtitanIntegration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Widgets Test PrintSupport Sql Svg)
qt_standard_project_setup()
enable_testing()
set(FMSTYLE_WITH_QTITAN ON CACHE BOOL "" FORCE)
set(FMSTYLE_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("../../externalLibs/waterColorStyle" watercolor)

# Compile the existing regression header against the same target usage requirements.
qt_add_executable(hc_qtitan_integration tstQtitanIntegration.cpp)
target_include_directories(hc_qtitan_integration PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(hc_qtitan_integration PRIVATE Qtitan::Grid Qt6::Test Fm::style)
add_test(NAME hc_qtitan_integration COMMAND hc_qtitan_integration
    -o "${CMAKE_CURRENT_BINARY_DIR}/hc-qtitan-$<CONFIG>.txt,txt")
set_tests_properties(hc_qtitan_integration PROPERTIES TIMEOUT 180)
get_target_property(QtQmake Qt6::qmake IMPORTED_LOCATION)
get_filename_component(QtBin "${QtQmake}" DIRECTORY)
set(TestEnvironment "QT_QPA_PLATFORM=offscreen" "PATH=${QtBin}\;$ENV{PATH}")
if(WIN32)
    list(APPEND TestEnvironment "QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
endif()
set_tests_properties(hc_qtitan_integration PROPERTIES ENVIRONMENT "${TestEnvironment}")
