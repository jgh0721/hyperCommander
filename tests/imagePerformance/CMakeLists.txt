cmake_minimum_required(VERSION 4.2)
project(HyperViewerImageComparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Gui Concurrent)
qt_standard_project_setup(REQUIRES 6.12)
get_filename_component(HC_ROOT "../.." ABSOLUTE)
set(HC_PERF_ROOT "${HC_ROOT}/_build/image-performance" CACHE PATH "Isolated comparison data")
set(HC_VENDOR "${HC_PERF_ROOT}/vendor")
set(ZIMG_ROOT "${HC_VENDOR}/zimg-release-3.0.6")
file(GLOB_RECURSE ZIMG_SOURCES CONFIGURE_DEPENDS "${ZIMG_ROOT}/src/zimg/*.cpp")
list(FILTER ZIMG_SOURCES EXCLUDE REGEX "/arm/")
qt_add_library(hc_perf_zimg STATIC ${ZIMG_SOURCES})
target_include_directories(hc_perf_zimg PUBLIC "${ZIMG_ROOT}/src/zimg/api" PRIVATE "${ZIMG_ROOT}/src/zimg")
target_compile_definitions(hc_perf_zimg PRIVATE ZIMG_X86 ZIMG_X86_AVX512 _CRT_SECURE_NO_WARNINGS NOMINMAX)
foreach(SOURCE IN LISTS ZIMG_SOURCES)
    if(SOURCE MATCHES "avx512")
        set_source_files_properties("${SOURCE}" PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
    elseif(SOURCE MATCHES "avx2")
        set_source_files_properties("${SOURCE}" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    elseif(SOURCE MATCHES "avx")
        set_source_files_properties("${SOURCE}" PROPERTIES COMPILE_OPTIONS "/arch:AVX")
    endif()
endforeach()
qt_add_executable(hc_image_comparison
    main.cpp gpu.cpp gpu.hpp avifCompare.cpp avifCompare.hpp stdafx.h
    "${HC_ROOT}/UniqueLibs/document/imageResampler.cpp"
    "${HC_PERF_ROOT}/reference.cpp")
target_include_directories(hc_image_comparison PRIVATE "${HC_ROOT}" "${HC_VENDOR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(hc_image_comparison PRIVATE "${HC_ROOT}/_build/image-codecs/runtime/include")
target_link_libraries(hc_image_comparison PRIVATE Qt6::Gui Qt6::Concurrent hc_perf_zimg windowscodecs ole32 d3d11 d3dcompiler dxgi psapi)
target_compile_definitions(hc_image_comparison PRIVATE NOMINMAX HC_IMAGE_CODEC_RUNTIME="${HC_ROOT}/_build/image-codecs/runtime/bin")
target_compile_options(hc_image_comparison PRIVATE /utf-8)
