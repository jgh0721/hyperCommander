cmake_minimum_required(VERSION 4.2)
get_filename_component(HC_UPDATER_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    # The recovery executable can also be configured without Qt or the GUI.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HC_UPDATER_REPO_ROOT}/CMakeLists.txt")
    file(READ "${HC_UPDATER_REPO_ROOT}/CMakeLists.txt" HC_UPDATER_ROOT_CMAKE)
    string(REGEX MATCH "set\\(HC_PRODUCT_VERSION \"([0-9.]+)\"\\)" HC_UPDATER_VERSION_MATCH "${HC_UPDATER_ROOT_CMAKE}")
    if(NOT HC_UPDATER_VERSION_MATCH)
        message(FATAL_ERROR "Cannot read the product version")
    endif()
    set(HC_PRODUCT_VERSION "${CMAKE_MATCH_1}")
    project(HyperCommanderUpdater VERSION ${HC_PRODUCT_VERSION} LANGUAGES CXX RC)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include("${HC_UPDATER_REPO_ROOT}/cmake/build.version.cmake")
    hc_split_version("${HC_PRODUCT_VERSION}" HC_VER_MAJOR HC_VER_MINOR HC_VER_PATCH HC_VER_TWEAK)
    set(HC_UPDATER_STANDALONE TRUE)
endif()
include("${HC_UPDATER_REPO_ROOT}/cmake/UpdateTrustPolicy.cmake")
hc_generate_update_trust_policy("${CMAKE_CURRENT_BINARY_DIR}/updateSigningPolicy.hpp")
# Shared trust and transaction code deliberately imports no Qt APIs. The helper
# must remain runnable when a Qt DLL is absent or an update was interrupted.
add_library(hc_update_core STATIC updateJson.cpp updateTrust.cpp updateProtocol.cpp updateTransaction.cpp)
target_precompile_headers(hc_update_core PRIVATE stdafx.h)
set_target_properties(hc_update_core PROPERTIES AUTOMOC OFF AUTOUIC OFF)
target_include_directories(hc_update_core PUBLIC ${HC_UPDATER_REPO_ROOT} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(hc_update_core PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(hc_update_core PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
if(MSVC)
    set_property(TARGET hc_update_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
target_link_libraries(hc_update_core PUBLIC wintrust crypt32 bcrypt version advapi32 PRIVATE user32)
add_executable(HyperCommanderUpdater WIN32 updaterMain.cpp)
target_precompile_headers(HyperCommanderUpdater PRIVATE stdafx.h)
set_target_properties(HyperCommanderUpdater PROPERTIES OUTPUT_NAME "HyperCommanderUpdater"
    AUTOMOC OFF AUTOUIC OFF HC_NATIVE_EXECUTABLE TRUE MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
hc_add_version_resource(HyperCommanderUpdater "HyperCommander Portable Updater")
target_link_libraries(HyperCommanderUpdater PRIVATE hc_update_core shell32 user32)
target_compile_options(HyperCommanderUpdater PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_sources(HyperCommanderUpdater PRIVATE "${HC_UPDATER_REPO_ROOT}/cmake/updateLongPaths.manifest")
if(MSVC)
    target_link_options(HyperCommanderUpdater PRIVATE "/NODEFAULTLIB:libucrt$<$<CONFIG:Debug>:d>" "/DEFAULTLIB:ucrt$<$<CONFIG:Debug>:d>")
endif()
if(NOT HC_UPDATER_STANDALONE)
    qt_add_library(hc_update_qt STATIC updateArchive.cpp updateArchive.hpp updateController.cpp updateController.hpp)
    target_precompile_headers(hc_update_qt PRIVATE stdafx.h)
    target_link_libraries(hc_update_qt PUBLIC hc_update_core Qt6::Widgets PRIVATE hc_sevenzip_backend Qt6::Network Qt6::Concurrent)
    target_compile_options(hc_update_qt PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
endif()
