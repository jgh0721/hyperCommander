# Production builds consume the checked-in offline asset set; npm is only needed to update it.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets.json" MARKDOWN_MANIFEST)
string(JSON MARKDOWN_ASSET_COUNT LENGTH "${MARKDOWN_MANIFEST}" files)
math(EXPR MARKDOWN_ASSET_LAST "${MARKDOWN_ASSET_COUNT} - 1")
foreach(INDEX RANGE 0 ${MARKDOWN_ASSET_LAST})
    string(JSON ASSET_PATH GET "${MARKDOWN_MANIFEST}" files ${INDEX} path)
    string(JSON ASSET_HASH GET "${MARKDOWN_MANIFEST}" files ${INDEX} sha256)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${ASSET_PATH}" ACTUAL_HASH)
    if(NOT ACTUAL_HASH STREQUAL ASSET_HASH)
        message(FATAL_ERROR "Markdown asset changed: ${ASSET_PATH}. Run npm ci and npm run build in tools/markdown.")
    endif()
endforeach()
find_package(Qt6 6.12.0 REQUIRED COMPONENTS WebEngineWidgets WebChannel Network)
qt_add_executable(hc-markdown-render WIN32 render/main.cpp render/stdafx.h)
set_target_properties(hc-markdown-render PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/Libs/Markdown")
target_precompile_headers(hc-markdown-render PRIVATE render/stdafx.h)
target_link_libraries(hc-markdown-render PRIVATE Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Network)
target_compile_options(hc-markdown-render PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
file(GLOB_RECURSE MARKDOWN_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/*")
qt_add_resources(hc-markdown-render "markdown_assets" PREFIX "/markdown" BASE "${CMAKE_CURRENT_SOURCE_DIR}/web" FILES ${MARKDOWN_ASSETS})
hc_add_version_resource(hc-markdown-render "Markdown renderer")
hc_enable_qt_deployment(hc-markdown-render PLUGINS)
add_custom_command(TARGET hc-markdown-render POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${CMAKE_BINARY_DIR}/$<CONFIG>/Licenses/Markdown"
    VERBATIM)
