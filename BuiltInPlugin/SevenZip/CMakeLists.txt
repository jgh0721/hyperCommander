enable_language(C)
set(HC_7ZIP_HASH 9cbde5099c6deb73691b0579063da5827522ccbbcba3f0020fd04e8c8c16c0d4)
set(HC_7ZIP_ARCHIVE "${CMAKE_BINARY_DIR}/_deps/7z2603-src.tar.xz")
set(HC_7ZIP_SOURCE "${CMAKE_BINARY_DIR}/_deps/7zip-26.03")
if(NOT EXISTS "${HC_7ZIP_ARCHIVE}")
    file(DOWNLOAD "https://www.7-zip.org/a/7z2603-src.tar.xz" "${HC_7ZIP_ARCHIVE}"
        EXPECTED_HASH "SHA256=${HC_7ZIP_HASH}" TLS_VERIFY ON)
endif()
file(SHA256 "${HC_7ZIP_ARCHIVE}" ACTUAL_HASH)
if(NOT ACTUAL_HASH STREQUAL HC_7ZIP_HASH)
    message(FATAL_ERROR "7-Zip source archive identity mismatch")
endif()
if(NOT EXISTS "${HC_7ZIP_SOURCE}/CPP/7zip/Bundles/Format7zF/Arc.mak")
    file(ARCHIVE_EXTRACT INPUT "${HC_7ZIP_ARCHIVE}" DESTINATION "${HC_7ZIP_SOURCE}")
endif()
# Full upstream bundle with portable C implementations of optional ASM objects.
set(COMMON_OBJS CPP/Common)
set(WIN_OBJS CPP/Windows)
set(7ZIP_COMMON_OBJS CPP/7zip/Common)
set(AR_OBJS CPP/7zip/Archive)
set(AR_COMMON_OBJS CPP/7zip/Archive/Common)
set(COMPRESS_OBJS CPP/7zip/Compress)
set(CRYPTO_OBJS CPP/7zip/Crypto)
set(C_OBJS C)
foreach(PAIR IN ITEMS "7Z|7z" "CAB|Cab" "CHM|Chm" "ISO|Iso" "NSIS|Nsis" "RAR|Rar" "TAR|Tar" "UDF|Udf" "WIM|Wim" "ZIP|Zip")
    string(REPLACE "|" ";" PARTS "${PAIR}")
    list(GET PARTS 0 KEY)
    list(GET PARTS 1 DIR)
    set(${KEY}_OBJS "CPP/7zip/Archive/${DIR}")
endforeach()
file(READ "${HC_7ZIP_SOURCE}/CPP/7zip/Bundles/Format7zF/Arc.mak" BUNDLE)
string(REPLACE "\\" "/" BUNDLE "${BUNDLE}")
string(REPLACE "\n" ";" BUNDLE "${BUNDLE}")
set(SOURCES)
foreach(LINE IN LISTS BUNDLE)
    if(LINE MATCHES "^([A-Z0-9_]+_OBJS) =")
        set(GROUP "${CMAKE_MATCH_1}")
    elseif(LINE MATCHES "[$]O/([A-Za-z0-9_]+)[.]obj")
        set(NAME "${CMAKE_MATCH_1}")
        set(EXT cpp)
        if(GROUP STREQUAL "C_OBJS")
            set(EXT c)
        endif()
        list(APPEND SOURCES "${HC_7ZIP_SOURCE}/${${GROUP}}/${NAME}.${EXT}")
    endif()
endforeach()
foreach(NAME IN ITEMS Aes AesOpt 7zCrc 7zCrcOpt XzCrc64 XzCrc64Opt LzFindOpt Sha1 Sha1Opt Sha256 Sha256Opt Sort)
    list(APPEND SOURCES "${HC_7ZIP_SOURCE}/C/${NAME}.c")
endforeach()
foreach(NAME IN ITEMS Sha1Prepare Sha256Prepare)
    list(APPEND SOURCES "${HC_7ZIP_SOURCE}/CPP/Common/${NAME}.cpp")
endforeach()
list(APPEND SOURCES
    "${HC_7ZIP_SOURCE}/CPP/7zip/Compress/CodecExports.cpp"
    "${HC_7ZIP_SOURCE}/CPP/7zip/Archive/ArchiveExports.cpp"
    "${HC_7ZIP_SOURCE}/CPP/7zip/Archive/DllExports2.cpp"
    "${HC_7ZIP_SOURCE}/CPP/7zip/Archive/Archive2.def")
list(REMOVE_DUPLICATES SOURCES)
qt_add_library(hc_sevenzip_core SHARED ${SOURCES})
set_target_properties(hc_sevenzip_core PROPERTIES AUTOMOC OFF AUTOUIC OFF OUTPUT_NAME 7z CXX_STANDARD 17)
target_compile_definitions(hc_sevenzip_core PRIVATE UNICODE _UNICODE Z7_EXTERNAL_CODECS)
target_link_libraries(hc_sevenzip_core PRIVATE oleaut32 advapi32 user32)
qt_add_library(hc_sevenzip_backend STATIC sevenZipBackend.cpp sevenZipBackend.hpp sevenZipUpdate.inl sevenZipMetadata.inl sevenZipMethods.inl sevenZipGroups.inl)
target_include_directories(hc_sevenzip_backend PRIVATE "${HC_7ZIP_SOURCE}/CPP")
target_link_libraries(hc_sevenzip_backend PUBLIC hc_plugin_contract PRIVATE Qt6::Core oleaut32)
target_compile_options(hc_sevenzip_backend PRIVATE $<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/utf-8>)
target_link_libraries(hc-plugin-host-x64 PRIVATE hc_sevenzip_backend)
set(PIN_HEADER "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/hcSevenZipManifest.hpp")
set(PACKAGE_COMMAND "${CMAKE_COMMAND}"
    "-DCORE=$<TARGET_FILE:hc_sevenzip_core>"
    "-DOUTPUT=$<TARGET_FILE_DIR:hc_sevenzip_core>/BuiltInPlugin/SevenZip"
    "-DHEADER=${PIN_HEADER}" "-DSOURCE=${HC_7ZIP_SOURCE}" "-DARCHIVE=${HC_7ZIP_ARCHIVE}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/GeneratePackage.cmake")
add_custom_command(OUTPUT "${PIN_HEADER}" COMMAND ${PACKAGE_COMMAND}
    DEPENDS hc_sevenzip_core GeneratePackage.cmake VERBATIM)
add_custom_target(hc_sevenzip_package COMMAND ${PACKAGE_COMMAND} DEPENDS "${PIN_HEADER}" VERBATIM)
qt_add_library(hc_builtin_sevenzip STATIC builtinSevenZip.cpp builtinSevenZip.hpp sevenZipJob.cpp sevenZipJob.hpp)
target_include_directories(hc_builtin_sevenzip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
target_link_libraries(hc_builtin_sevenzip PUBLIC hc_plugin_process)
target_compile_options(hc_builtin_sevenzip PRIVATE $<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/utf-8>)
add_dependencies(hc_builtin_sevenzip hc_sevenzip_package)
